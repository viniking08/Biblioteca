DROP DATABASE IF EXISTS biblioteca;
CREATE DATABASE biblioteca;
USE biblioteca;

CREATE TABLE autores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_completo VARCHAR(150),
    nacionalidade VARCHAR(80),
    data_nascimento DATE
);

CREATE TABLE livros (
    id INT PRIMARY KEY,
    titulo VARCHAR(200),
    isbn VARCHAR(20) UNIQUE,
    ano_publicacao INT,
    numero_paginas INT,
    sinopse TEXT
);

CREATE TABLE generos (
    id INT PRIMARY KEY,
    nome VARCHAR(80)
);

CREATE TABLE usuarios (
    id INT PRIMARY KEY,
    nome_completo VARCHAR(150),
    cpf VARCHAR(20),
    email VARCHAR(150),
    telefone VARCHAR(20),
    data_nascimento DATE
);

CREATE TABLE emprestimos (
    id INT PRIMARY KEY,
    data_emprestimos DATE,
    data_devolucao DATE,
    usuarios_id INT,
    livros_id INT,

    FOREIGN KEY (usuarios_id) REFERENCES usuarios(id),
    FOREIGN KEY (livros_id) REFERENCES livros(id)
);

CREATE TABLE autores_has_livros (
    autores_id INT,
    livros_id INT,

    PRIMARY KEY (autores_id, livros_id),
    FOREIGN KEY (autores_id) REFERENCES autores(id),
    FOREIGN KEY (livros_id) REFERENCES livros(id)
);

CREATE TABLE livros_has_generos (
    livros_id INT,
    generos_id INT,

    PRIMARY KEY (livros_id, generos_id),
    FOREIGN KEY (livros_id) REFERENCES livros(id),
    FOREIGN KEY (generos_id) REFERENCES generos(id)
);
